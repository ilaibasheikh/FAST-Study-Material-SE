-- 1. Department Table
CREATE TABLE Department (
    DeptID INT PRIMARY KEY,
    DeptName VARCHAR(100)
);

-- 2. Faculty Table
CREATE TABLE Faculty (
    FacultyID INT PRIMARY KEY,
    FacultyName VARCHAR(100),
    DeptID INT FOREIGN KEY REFERENCES Department(DeptID)
);

-- 3. Student Table
CREATE TABLE Student (
    StudentID INT PRIMARY KEY,
    StudentName VARCHAR(100),
    DeptID INT FOREIGN KEY REFERENCES Department(DeptID),
    EnrollmentYear INT
);

-- 4. Course Table
CREATE TABLE Course (
    CourseID INT PRIMARY KEY,
    CourseName VARCHAR(100),
    Credits INT,
    DeptID INT FOREIGN KEY REFERENCES Department(DeptID)
);

-- 5. CourseOffering Table
CREATE TABLE CourseOffering (
    OfferingID INT PRIMARY KEY,
    CourseID INT FOREIGN KEY REFERENCES Course(CourseID),
    FacultyID INT FOREIGN KEY REFERENCES Faculty(FacultyID),
    Semester VARCHAR(10),
    Year INT
);

-- 6. Enrollment Table
CREATE TABLE Enrollment (
    EnrollmentID INT PRIMARY KEY,
    StudentID INT FOREIGN KEY REFERENCES Student(StudentID),
    OfferingID INT FOREIGN KEY REFERENCES CourseOffering(OfferingID),
    Grade CHAR(2)
);

-- 7. Classroom Table
CREATE TABLE Classroom (
    RoomID INT PRIMARY KEY,
    Building VARCHAR(50),
    Capacity INT
);

-- 8. Timetable Table
CREATE TABLE Timetable (
    ScheduleID INT PRIMARY KEY,
    OfferingID INT FOREIGN KEY REFERENCES CourseOffering(OfferingID),
    RoomID INT FOREIGN KEY REFERENCES Classroom(RoomID),
    DayOfWeek VARCHAR(10),
    StartTime TIME,
    EndTime TIME
);

-- 9. Attendance Table
CREATE TABLE Attendance (
    AttendanceID INT PRIMARY KEY,
    EnrollmentID INT FOREIGN KEY REFERENCES Enrollment(EnrollmentID),
    AttendanceDate DATE,
    Status VARCHAR(10)  -- Present/Absent
);

-- 10. Admin Table
CREATE TABLE Admin (
    AdminID INT PRIMARY KEY,
    AdminName VARCHAR(100),
    DeptID INT FOREIGN KEY REFERENCES Department(DeptID)
);
